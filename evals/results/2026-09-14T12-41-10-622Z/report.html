\section{Formal Security}
\label{sec:security}

\subsection{The Ideal Functionality $\mathcal{F}_{\mathsf{encmp}}$}
\label{sec:functionality}

Let $\mathcal{P}=\{P_1,\dots,P_n\}$ be the sequencer committee with $n=16$ and
reconstruction threshold $t=6$, and let $f=5$ bound the number of corruptions.
A bundle is a pair $b=(\mathsf{hdr},m)$ in which
$\mathsf{hdr}=(\mathsf{sender},\mathsf{nonce},\mathsf{gasLimit},\mathsf{maxFeePerGas})$
is the \emph{admission header} and $m=(\mathsf{to},\mathsf{value},\mathsf{calldata})$
is the \emph{payload}. Admission control is a public predicate
$\mathsf{Adm}(\mathsf{hdr},\mathsf{st})$ over the header and the current chain
state $\mathsf{st}$: a mempool that cannot evaluate $\mathsf{Adm}$ before ordering
cannot reject bundles that are guaranteed to revert, and is trivially floodable.
We therefore model $\mathsf{hdr}$ as a \emph{public input} of the mempool
interface rather than as a secret, and state confidentiality for the payload
$m$, which is what ordering-dependent value extraction acts on.
Inferences a searcher may draw from headers alone (e.g.\ fee-implied urgency)
are outside the guarantee proved here.

\subsection{Security Theorem}
\label{sec:theorem}

We assume the threshold scheme $\mathsf{TPKE}=(\mathsf{Setup},\mathsf{Enc},
\mathsf{ShareDec},\mathsf{Combine})$ satisfies two properties.
\emph{(i) Threshold IND-CPA:} no PPT adversary holding $\{\mathit{sk}_i\}_{i\in\mathcal{C}}$
for any $\mathcal{C}$ with $|\mathcal{C}|\le t-1=5$ distinguishes
$\mathsf{Enc}(\mathit{cpk},m_0)$ from $\mathsf{Enc}(\mathit{cpk},m_1)$ for
$|m_0|=|m_1|$; its advantage is $\mathsf{Adv}^{\mathsf{tcpa}}_{\mathsf{TPKE},f}(\lambda)$.
\emph{(ii) Simulatable decryption shares:} there is a deterministic polynomial-time
$\mathsf{SimShare}$ such that for every $\mathit{ct}$, every $m$, and every
$\mathcal{C}$ with $|\mathcal{C}|\le f$,
$\mathsf{SimShare}(\mathit{cpk},\mathit{ct},m,\{\mathit{sk}_i\}_{i\in\mathcal{C}})$
outputs shares $\{\sigma_j\}_{j\notin\mathcal{C}}$ identically distributed to
$\{\mathsf{ShareDec}(\mathit{sk}_j,\mathit{ct})\}_{j\notin\mathcal{C}}$ whenever
$\mathit{ct}\in\mathsf{Enc}(\mathit{cpk},m)$. Property~(ii) holds perfectly for
the ElGamal-style instantiation of Section~\ref{sec:construction}: the corrupted
shares plus the target plaintext fix the degree-$(t-1)$ sharing polynomial in the
exponent, determining every honest share by Lagrange interpolation.
We work in the $\mathcal{F}_{\mathsf{bc}}$-hybrid model, where
$\mathcal{F}_{\mathsf{bc}}$ is authenticated broadcast; this discharges the
ordering certificate's signatures, which play no role in confidentiality.

\begin{theorem}[Payload confidentiality]
\label{thm:encmp}
Let $\mathcal{A}$ be a PPT adversary that statically and semi-honestly corrupts
$\mathcal{C}\subset[n]$ with $|\mathcal{C}|\le f=5$ and additionally observes
every message sent over $\mathcal{F}_{\mathsf{bc}}$. Then there is a PPT
simulator $\mathcal{S}$ such that for every PPT environment $\mathcal{Z}$
submitting at most $q=\mathrm{poly}(\lambda)$ bundles,
\[
\Big|\Pr\big[\mathsf{REAL}_{\Pi_{\mathsf{encmp}},\mathcal{A},\mathcal{Z}}(\lambda)=1\big]
-\Pr\big[\mathsf{IDEAL}_{\mathcal{F}_{\mathsf{encmp}},\mathcal{S},\mathcal{Z}}(\lambda)=1\big]\Big|
\;\le\; q\cdot\mathsf{Adv}^{\mathsf{tcpa}}_{\mathsf{TPKE},f}(\lambda).
\]
\end{theorem}

\noindent
The global-observer searcher needs no separate treatment: a corrupted committee
member already sees every broadcast message and additionally holds a key share,
so folding the searcher into $\mathcal{A}$ yields a strictly stronger adversary.

\begin{corollary}[No payload-dependent front-running]
\label{cor:frontrun}
Any value $\mathcal{A}$ outputs before $(\textsc{ordered},\mathit{sid},\pi)$ is,
up to $q\cdot\mathsf{Adv}^{\mathsf{tcpa}}_{\mathsf{TPKE},f}(\lambda)$, a function
of the headers and payload lengths alone. In particular the committed order
$\pi$ is independent of $\{m_{\mathit{bid}}\}$ conditioned on
$\{(\mathsf{hdr}_{\mathit{bid}},|m_{\mathit{bid}}|)\}$.
\end{corollary}

\subsection{Proof}
\label{sec:proof}

\begin{proof}[Proof of Theorem~\ref{thm:encmp}]
\textbf{Simulator.} $\mathcal{S}$ runs $\mathcal{A}$ internally and proceeds as
follows.
\begin{enumerate}\itemsep1pt \parskip0pt \topsep2pt
  \item \emph{Setup.} $\mathcal{S}$ runs $(\mathit{cpk},\{\mathit{sk}_i\}_{i\in[n]})
  \gets\mathsf{Setup}(1^\lambda,n,t)$ and hands $\mathit{cpk}$ and
  $\{\mathit{sk}_i\}_{i\in\mathcal{C}}$ to $\mathcal{A}$. If the deployment uses a
  DKG instead of a dealer, $\mathcal{S}$ invokes its simulator, which exists since
  $|\mathcal{C}|\le f<n/2$.
  \item \emph{Submission.} On $(\textsc{submitted},\mathit{sid},\mathit{bid},
  \mathsf{hdr},\ell)$ from $\mathcal{F}_{\mathsf{encmp}}$, $\mathcal{S}$ sets
  $\mathit{ct}_{\mathit{bid}}\gets\mathsf{Enc}(\mathit{cpk},0^{\ell})$ and delivers
  $(\mathit{bid},\mathsf{hdr},\mathit{ct}_{\mathit{bid}})$ to $\mathcal{A}$ on
  behalf of $\mathcal{F}_{\mathsf{bc}}$. The corrupted members then evaluate
  $\mathsf{Adm}(\mathsf{hdr},\mathsf{st})$ exactly as in the real protocol; this is
  the step for which $\mathcal{S}$ requires $\mathsf{hdr}$, and it is why
  $\mathsf{hdr}$ is an output of $\mathcal{F}_{\mathsf{encmp}}$ rather than a
  value the functionality withholds.
  \item \emph{Ordering.} $\mathcal{S}$ simulates the leader's proposal $\pi$ over
  the pending $(\mathit{bid},\mathsf{hdr},\mathit{ct})$ triples and the honest
  members' approvals, all of which are functions of values it already holds;
  corrupted members approve according to the protocol, since $\mathcal{A}$ is
  semi-honest. $\mathcal{S}$ then sends $(\textsc{order},\mathit{sid},\pi)$ to
  $\mathcal{F}_{\mathsf{encmp}}$.
  \item \emph{Opening.} On
  $(\textsc{opened},\mathit{sid},\pi,\{(\mathit{bid},m_{\mathit{bid}})\})$,
  $\mathcal{S}$ computes
  $\{\sigma_j\}_{j\notin\mathcal{C}}\gets\mathsf{SimShare}(\mathit{cpk},
  \mathit{ct}_{\mathit{bid}},m_{\mathit{bid}},\{\mathit{sk}_i\}_{i\in\mathcal{C}})$
  for each $\mathit{bid}\in\pi$ and broadcasts them as the honest members' shares.
  Corrupted members produce their own shares honestly. $\mathcal{S}$ outputs
  whatever $\mathcal{A}$ outputs.
\end{enumerate}

\noindent\textbf{Hybrids.} Let $\mathsf{H}_0$ be
$\mathsf{REAL}_{\Pi_{\mathsf{encmp}},\mathcal{A},\mathcal{Z}}(\lambda)$.

\emph{$\mathsf{H}_1$: simulated setup.} Key material is generated by
$\mathcal{S}$ as in step~1. With a dealer the distribution is identical; with a
DKG it is indistinguishable by the DKG simulator, contributing a negligible term
absorbed below.

\emph{$\mathsf{H}_2$: simulated honest shares.} Every honest decryption share is
produced by $\mathsf{SimShare}$ on the true plaintext instead of by
$\mathsf{ShareDec}$. Ciphertexts are still encryptions of the true payloads, so
property~(ii) applies and $\mathsf{H}_2\equiv\mathsf{H}_1$. This step must precede
the next one: once ciphertexts encrypt $0^{\ell}$, honest shares computed by
$\mathsf{ShareDec}$ would reconstruct $0^{\ell}$ rather than the payload the
environment expects to see executed.

\emph{$\mathsf{H}_3^{k}$ for $k=0,\dots,q$: ciphertext replacement.} The first
$k$ submitted bundles are encrypted as $\mathsf{Enc}(\mathit{cpk},0^{\ell})$ and
the remaining $q-k$ honestly; $\mathsf{H}_3^{0}=\mathsf{H}_2$ and
$\mathsf{H}_3^{q}=\mathsf{IDEAL}_{\mathcal{F}_{\mathsf{encmp}},\mathcal{S},\mathcal{Z}}(\lambda)$.

\begin{claim}
$\big|\Pr[\mathsf{H}_3^{k-1}=1]-\Pr[\mathsf{H}_3^{k}=1]\big|
\le\mathsf{Adv}^{\mathsf{tcpa}}_{\mathsf{TPKE},f}(\lambda)$ for each $k\in[q]$.
\end{claim}
\begin{proof}[Proof of Claim]
Build $\mathcal{B}$ against threshold IND-CPA. $\mathcal{B}$ receives
$\mathit{cpk}$ and $\{\mathit{sk}_i\}_{i\in\mathcal{C}}$ from its challenger and
forwards them to $\mathcal{A}$. It runs $\mathsf{H}_3^{k-1}$, except that on the
$k$-th submission $(\mathsf{hdr},m_k)$ it submits $(m_k,0^{|m_k|})$ as its
challenge pair and uses the returned $\mathit{ct}^\ast$ for that bundle. Note
$\mathcal{B}$ never needs $\mathit{sk}_j$ for $j\notin\mathcal{C}$: by
$\mathsf{H}_2$, all honest shares, including those for $\mathit{ct}^\ast$, come
from $\mathsf{SimShare}$ on $(\mathit{ct}^\ast,m_k,\{\mathit{sk}_i\}_{i\in\mathcal{C}})$,
and $\mathcal{B}$ knows $m_k$ because $\mathcal{Z}$ chose it. When the challenger
encrypts $m_k$ the view is $\mathsf{H}_3^{k-1}$; when it encrypts $0^{|m_k|}$ it
is $\mathsf{H}_3^{k}$. $\mathcal{B}$ echoes the distinguisher's bit.
\end{proof}

\noindent
Summing over $k$ and applying transitivity gives the bound of
Theorem~\ref{thm:encmp}. Since $|\mathcal{C}|\le 5<t=6$, $\mathcal{A}$ holds too
few shares to reconstruct any pending payload on its own, and the protocol
releases honest shares only after the ordering certificate is formed; hence no
decryption-share information about an uncommitted bundle appears in any hybrid.
\end{proof}

\section{Formal Security}
\label{sec:security}

\subsection{Ideal Functionality}
\label{subsec:func}

We work in the standalone simulation paradigm with \emph{static, semi-honest}
corruptions. A bundle is a pair $b=(\mathsf{hdr},m)$: the header
$\mathsf{hdr}=(\mathsf{sender},\mathsf{nonce},\mathsf{gasLimit},\mathsf{maxFeePerGas})$
carries the fields the committee must read in order to admit $b$ into the pool, and
the payload $m=(\mathsf{to},\mathsf{value},\mathsf{calldata})$ carries the transaction
content, which the protocol transports as $\mathit{ct}\gets\mathsf{TPKE.Enc}(\mathit{cpk},m)$.
Admission is the public predicate $\mathsf{adm}(\mathit{st},\mathsf{hdr})$ evaluated against
chain state $\mathit{st}$ (the sender's balance covers
$\mathsf{gasLimit}\cdot\mathsf{maxFeePerGas}$; the nonce is contiguous), and
$\mathsf{Valid}(\pi,P)$ denotes the ordering rule of \S\ref{sec:ordering}.
Ciphertext length is a public function $\ell(|m|)$ of payload length.

Headers travel in the clear---the committee cannot evaluate $\mathsf{adm}$
otherwise---so they are part of what the functionality hands to the adversary. We
define the per-bundle leakage as
\[
  \mathcal{L}(b)\;=\;\bigl(\mathsf{hdr},\,|\mathit{ct}|\bigr),
\]
and observe that $\mathcal{L}$ is \emph{independent of $m$}: what
$\mathcal{F}_{\mathsf{encmp}}$ withholds until the order is fixed is exactly the
transaction content.

\subsection{Security Theorem}
\label{subsec:thm}

\begin{theorem}
\label{thm:main}
Let $\mathsf{TPKE}$ be a $(6,16)$-threshold public-key encryption scheme that is
IND-CPA secure under static corruption of at most $5$ key shares. For every static,
semi-honest PPT adversary $\mathcal{A}$ corrupting a set $C$ of at most $5$ committee
nodes together with a searcher that observes every public message, and every
environment $\mathcal{Z}$, there exists a PPT simulator $\mathcal{S}$ such that
\[
  \mathrm{REAL}_{\Pi,\mathcal{A},\mathcal{Z}}(\lambda)
  \;\approx_c\;
  \mathrm{IDEAL}_{\mathcal{F}_{\mathsf{encmp}},\mathcal{S},\mathcal{Z}}(\lambda).
\]
Concretely, for an execution carrying $Q$ bundles submitted by honest users, there is a
PPT $\mathcal{B}$ with
$\mathbf{Adv}^{\mathrm{dist}}_{\Pi}(\mathcal{A},\mathcal{Z})
 \le Q\cdot\mathbf{Adv}^{\mathrm{ind\text{-}cpa}}_{\mathsf{TPKE}}(\mathcal{B})$.
\end{theorem}

\begin{corollary}[No content-dependent front-running]
\label{cor:frontrun}
Fix any header vector $\vec{\mathsf{hdr}}$ and any two payload vectors
$\vec{m},\vec{m}'$ with $\ell(|m_i|)=\ell(|m_i'|)$ for all $i$. The distribution of the
committed order $\pi$ is computationally independent of which vector was submitted.
Hence no strategy of $\mathcal{A}$ can place a transaction relative to a victim's on
the basis of that victim's $(\mathsf{to},\mathsf{value},\mathsf{calldata})$.
\end{corollary}

\subsection{Proof}
\label{subsec:proof}

\begin{proof}[Proof of Theorem~\ref{thm:main}]
\textbf{The simulator.} $\mathcal{S}$ runs $\mathsf{TPKE.KeyGen}$, publishes $\mathit{cpk}$,
and hands $\{\mathit{sk}_j\}_{j\in C}$ to $\mathcal{A}$ (the corrupted set is fixed before
execution). It then maintains a simulated transcript as follows.
\emph{(i) Submission.} On $(\textsc{leak},\mathit{bid},\mathsf{hdr},\ell)$ it sets
$\widetilde{\mathit{ct}}\gets\mathsf{TPKE.Enc}(\mathit{cpk},0^{\ell})$ and delivers the
bundle $(\mathsf{hdr},\widetilde{\mathit{ct}})$ to $\mathcal{A}$ as the committee-wide
broadcast, both in $\mathcal{A}$'s role as a corrupted node and in its role as searcher.
\emph{(ii) Ordering.} $\mathcal{S}$ runs the honest leader code on the simulated pool if
the leader is honest, and otherwise reads $\pi$ off $\mathcal{A}$'s output; it signs on
behalf of honest nodes with its own keys and sends $(\textsc{order},\pi)$ to
$\mathcal{F}_{\mathsf{encmp}}$.
\emph{(iii) Decryption.} $\mathcal{S}$ sends $(\textsc{release})$, receives
$\{(\mathit{bid},\mathsf{hdr},m)\}_{\mathit{bid}\in\pi}$, and for each simulated
$\widetilde{\mathit{ct}}$ produces the honest nodes' decryption shares as
$\mathsf{ShareSim}(\mathit{cpk},\{\mathit{sk}_j\}_{j\in C},\widetilde{\mathit{ct}},m)$,
defined below. Corrupted nodes emit their own shares from $\mathit{sk}_j$, which
$\mathcal{S}$ needs not touch since $\mathcal{A}$ is semi-honest.

\textbf{$\mathsf{ShareSim}$.} The secret key is Shamir-shared along a polynomial $f$ of
degree $5$ with $f(0)=\mathit{sk}$, and node $i$'s share for a ciphertext with ephemeral
element $u$ is $u^{f(i)}$. The adversary holds $|C|\le 5$ evaluations, and the target
value $u^{f(0)}$ is pinned down by the released plaintext $m$. Six points in the exponent
determine a degree-$5$ polynomial, so $\mathsf{ShareSim}$ recovers every honest share by
Lagrange interpolation in the exponent. Crucially this uses only $(\mathit{ct},m)$ and the
corrupted shares, never $\mathit{sk}$, and never requires $\mathit{ct}$ to actually decrypt
to $m$. The corruption bound $5=t-1$ is what makes the shares \emph{uniquely} determined,
and hence this step exact rather than approximate.

\textbf{Hybrids.} $\mathbf{H}_0$ is the real execution.

$\mathbf{H}_1$ replaces the honest nodes' decryption shares by $\mathsf{ShareSim}$ output.
By the preceding paragraph the two share vectors are identical, so
$\mathbf{H}_1\equiv\mathbf{H}_0$.

$\mathbf{H}_2^{(i)}$, for $i=0,\dots,Q$, additionally replaces the first $i$ honest
bundles' ciphertexts by $\mathsf{Enc}(\mathit{cpk},0^{\ell})$, with
$\mathbf{H}_2^{(0)}=\mathbf{H}_1$. A distinguisher between $\mathbf{H}_2^{(i-1)}$ and
$\mathbf{H}_2^{(i)}$ yields $\mathcal{B}$ against threshold IND-CPA: $\mathcal{B}$ receives

simulates everything else as $\mathcal{S}$ does. Note that $\mathcal{B}$ never decrypts
$\mathit{ct}^{*}$: the shares opening it to $m_i$ come from $\mathsf{ShareSim}$, which needs
only $m_i$---a message $\mathcal{B}$ itself chose---and the corrupted key shares. Hence
$\mathcal{B}$ is a legal IND-CPA adversary, and it is precisely the ordering of the two
hops that makes this so: had the ciphertexts been swapped before the shares, $\mathcal{B}$
would have had to force a decryption it cannot compute.

Finally $\mathbf{H}_2^{(Q)}$ is the ideal execution, since every message in it is generated
from $\{\mathcal{L}(b)\}$ before ordering and from $\mathcal{F}_{\mathsf{encmp}}$'s output
after it, which is exactly $\mathcal{S}$'s interface. Summing over the $Q$ hops gives the
stated bound. Corollary~\ref{cor:frontrun} follows because $\pi$ is fixed in
$\mathbf{H}_2^{(Q)}$ before any payload enters the view.
\end{proof}

## 按 `F_encmp` 只泄漏 `|ct|` + 条数，这个定理是假的，证不出来

不是"不够严谨"的程度，是 simulator 卡死的程度：

real 世界里敌手的 view 包含 16 个节点收到的广播消息，也就是 `(sender, nonce, gasLimit, maxFeePerGas, ct)`——那 5 个腐化节点直接收到明文 header，那个全网 searcher 也看得到。ideal 世界里 simulator 从 `F_encmp` 只拿到 `|ct|` 和条数，**它没有任何地方能变出真实的 sender 和 nonce**。环境 Z 只要提交一笔自己知道 sender 的交易，再看敌手 view 里那个字段对不对，就能以优势 1 区分。所以 hybrid 走不到最后一步，中间那步会断。

\begin{remark}
The header is leaked at \textsc{submit} by design, not by accident: the
committee cannot admit a bundle without checking that $\mathsf{sender}$ can
fund $\mathsf{gasLimit}\cdot\mathsf{maxFeePerGas}$ and that $\mathsf{nonce}$ is
contiguous, and an encrypted mempool that skips this check is trivially
floodable with ciphertexts that decrypt to reverting transactions. Since
$\mathsf{sender}$ and $\mathsf{nonce}$ are in any case recoverable from chain
state, the security claim we make---and the one that matters for front-running
---is over $\pld$.
\end{remark}

\subsection{Security Theorem}

We require of the threshold scheme, besides threshold IND-CPA security, that
partial decryptions be simulatable: there is a PPT $\mathsf{ShareSim}$ that, on
input a ciphertext $\mathit{ct}$, a target plaintext $m$, and the key shares
$\{\mathit{sk}_j\}_{j\in C}$ of any $|C|\le t-1$ parties, outputs shares
$\{d_i\}_{i\notin C}$ distributed exactly as honest partial decryptions that
combine to $m$. For Shamir-shared ElGamal-style TPKE with $t=6$ and $|C|=5$
this holds \emph{perfectly} and without the master secret: the five corrupt
shares together with the combined value $m$ impose six constraints on a
degree-$5$ polynomial, so the eleven honest shares are uniquely determined and
computable by Lagrange interpolation in the exponent.

\begin{theorem}\label{thm:main}
Let $\mathsf{TPKE}$ be $(16,6)$-threshold IND-CPA secure with simulatable
partial decryption. Then $\Prot$ securely realises $\Fencmp$ in the
$(\mathcal{F}_{\mathsf{bc}},\mathcal{F}_{\mathsf{dkg}})$-hybrid model against
every static, semi-honest PPT adversary that corrupts a set $C$ of committee
members with $|C|\le 5$ and additionally observes every broadcast message.
Concretely, for every such $\Adv$ there is a PPT $\Sm$ such that for every
environment $\mathcal{Z}$ issuing at most $B$ submissions,
\[
\bigl|\Pr[\mathrm{REAL}_{\Prot,\Adv,\mathcal{Z}}(\lambda)=1]
-\Pr[\mathrm{IDEAL}_{\Fencmp,\Sm,\mathcal{Z}}(\lambda)=1]\bigr|
\;\le\; B\cdot\mathsf{Adv}^{\text{t-ind-cpa}}_{\mathsf{TPKE}}(\lambda).
\]
\end{theorem}

\begin{corollary}[No pre-ordering front-running]
The adversary's view up to the \textsc{ordered} message of an epoch is a
function of $(\hdr_i,|\pld_i|)_i$ and its own randomness alone; it is
independent of every payload in the batch. Hence no PPT strategy can order a
transaction of its own against an undisclosed $(\mathsf{to},\mathsf{value},
\mathsf{calldata})$ with better than negligible advantage over the same
strategy run on the headers alone.
\end{corollary}

\subsection{Proof}

\begin{proof}
\textbf{Simulator.} $\Sm$ runs $\Adv$ internally and emulates
$\mathcal{F}_{\mathsf{dkg}}$, handing each $j\in C$ its share
$\mathit{sk}_j$ and publishing $\mathit{cpk}$.
On $(\textsc{submitted},\mathit{mid},(\hdr,\ell))$ from $\Fencmp$, it samples
$\widetilde{\mathit{ct}}\leftarrow\mathsf{TPKE}.\mathsf{Enc}(\mathit{cpk},0^{\ell})$
and delivers the broadcast $(\hdr,\widetilde{\mathit{ct}})$ to every corrupt
node, exactly as an honest submitter would. It runs the leader's ordering rule
on the headers it holds, emulates the honest nodes' signatures on the
resulting $\vec{B}$, and sends $(\textsc{order},\vec{B})$ to $\Fencmp$.
On $(\textsc{reveal},\vec{B},(\pld_{\mathit{mid}})_{\mathit{mid}})$ it computes,
for each $\mathit{mid}\in\vec{B}$,
$\{d_i\}_{i\notin C}\leftarrow\mathsf{ShareSim}(\widetilde{\mathit{ct}}_{\mathit{mid}},
\pld_{\mathit{mid}},\{\mathit{sk}_j\}_{j\in C})$
and broadcasts them on behalf of the honest nodes. $\Sm$ outputs $\Adv$'s
output.

\textbf{Hybrids.} Let $\mathsf{H}_0$ be the real execution.

\emph{$\mathsf{H}_1$: simulated partial decryptions.} Honest nodes' decryption
shares are produced by $\mathsf{ShareSim}$ on the true plaintext instead of
with $\mathit{sk}_i$. Since $|C|=5=t-1$, the corrupt shares and the combined
plaintext determine the honest shares uniquely, so $\mathsf{H}_1\equiv
\mathsf{H}_0$: the change is syntactic and the distributions are identical.
This step is what lets the reduction below answer the reveal phase without the
master secret.

\emph{$\mathsf{H}_{1,i}$, $0\le i\le B$: ciphertext replacement.} In
$\mathsf{H}_{1,i}$ the first $i$ submissions are broadcast as
$\mathsf{Enc}(\mathit{cpk},0^{|\pld|})$ and the remaining $B-i$ as
$\mathsf{Enc}(\mathit{cpk},\pld)$; headers are unchanged throughout. By
construction $\mathsf{H}_{1,0}=\mathsf{H}_1$.

\emph{Claim: $\mathsf{H}_{1,i-1}\approx_c\mathsf{H}_{1,i}$.} Given a
distinguisher, build $\mathcal{R}$ against threshold IND-CPA. $\mathcal{R}$
receives $\mathit{cpk}$ and the corrupt key shares
$\{\mathit{sk}_j\}_{j\in C}$ from the challenger---admissible since
$|C|\le t-1$---and forwards them to $\Adv$ as the $\mathcal{F}_{\mathsf{dkg}}$
output. It runs $\mathcal{Z}$, encrypting submissions $<i$ to zero and $>i$
honestly. For the $i$-th it submits $(\pld_i,0^{|\pld_i|})$ and broadcasts the
challenge ciphertext. At reveal time $\mathcal{R}$ answers with
$\mathsf{ShareSim}(\mathit{ct}^{*},\pld_i,\{\mathit{sk}_j\}_{j\in C})$, which
needs only the corrupt shares and $\pld_i$; note $\mathcal{R}$ never queries a
decryption oracle on $\mathit{ct}^{*}$, so the reduction is admissible.
$\mathcal{R}$ outputs $\mathcal{Z}$'s bit, and its advantage equals the
distinguishing gap. Summing over $i=1,\dots,B$ gives the stated bound.

\emph{$\mathsf{H}_{1,B}\equiv\mathrm{IDEAL}$.} In $\mathsf{H}_{1,B}$ every
broadcast ciphertext encrypts zeros, headers come from $\Leak$, the order is
the leader's rule applied to those headers, and the shares are produced by
$\mathsf{ShareSim}$ from payloads first available at reveal time. This is
precisely $\Sm$ interacting with $\Fencmp$. Semi-honest, static corruption
means no further deviation must be simulated, and the environment's remaining
inputs are identically distributed in the two worlds.

Chaining the steps yields Theorem~\ref{thm:main}. The corollary follows because
$\mathsf{H}_{1,B}$ restricted to the prefix before \textsc{ordered} is
syntactically independent of $(\pld_i)_i$.
\end{proof}

\subsection{Security Theorem}

We rely on three assumptions on the primitives. \textbf{(A1)} $\mathsf{TPKE}$
is IND-CPA secure against an adversary statically corrupting up to $t = 5$
of the $n = 16$ key shares. \textbf{(A2)} $\mathsf{TPKE}$ has
\emph{simulatable decryption shares}: there is a PPT $\mathsf{ShareSim}$
such that for every ciphertext $\ct$, every $\tx$ with
$\mathsf{Dec}(\ct) = \tx$, and every $I \subseteq [16]$ with $|I| \le 5$,
the output of $\mathsf{ShareSim}(\ct, \tx, \{sk_i\}_{i \in I})$ is
statistically indistinguishable from the honest parties' shares
$\{\mathsf{ShareDec}(sk_j, \ct)\}_{j \notin I}$. \textbf{(A3)} the committee
signature scheme is EUF-CMA.

\begin{theorem}
\label{thm:main}
Under \emph{(A1)--(A3)}, \Pencmp{} UC-realises \Fencmp{} with leakage
$\leak(\hdr,\tx) = (\hdr, |\tx|)$ in the presence of a static, semi-honest
adversary corrupting at most $5$ committee nodes together with an arbitrary
number of external observers. Concretely, for every PPT \Adv{} there is a
PPT \Sim{} such that for every PPT \Env{} and every batch of $m$ bundles,
\[
  \Bigl|\Pr[\mathrm{EXEC}_{\Pencmp,\Adv,\Env} = 1]
       - \Pr[\mathrm{EXEC}_{\Fencmp,\Sim,\Env} = 1]\Bigr|
  \;\le\; \epsilon_{\mathrm{sig}} + m \cdot \epsilon_{\mathrm{cpa}}
       + \mathrm{negl}(\lambda).
\]
\end{theorem}

\begin{proof}[Proof sketch]
\emph{Simulator.} \Sim{} runs $\mathsf{TPKE.Setup}$ itself, publishes
$\mathsf{cpk}$, and hands \Adv{} the key shares of the (statically fixed)
corrupt set $I$, $|I| \le 5$; it retains all $16$ shares. On
$(\textsc{submitted}, sid, \beta, (\hdr, \ell))$ from \Fencmp{} it forms the
dummy bundle $\tilde{b}_\beta = (\hdr, \tilde{\ct}_\beta)$ with
$\tilde{\ct}_\beta \gets \mathsf{Enc}(\mathsf{cpk}, 0^{\ell})$ and delivers
it to \Adv{} as the committee-wide broadcast. Bundles originating from
corrupt users are extracted by decrypting with the retained shares and
replayed to \Fencmp{} on that user's behalf. If the leader is honest, \Sim{}
picks $\pi$ as the protocol prescribes; if the leader is corrupt, \Sim{}
takes the $\pi$ and the committee signature output by \Adv, verifies the
signature, and sends $(\textsc{order}, sid, \pi)$ to \Fencmp. Only then does
\Sim{} send $(\textsc{open}, sid)$, receive $\{\tx_\beta\}$, and produce the
honest nodes' decryption shares for each $\tilde{\ct}_\beta$ as
$\mathsf{ShareSim}(\tilde{\ct}_\beta, \tx_\beta, \{sk_i\}_{i \in I})$.

\emph{Hybrids.} $\mathbf{H}_0$ is the real execution.

$\mathbf{H}_1$ aborts if \Adv{} ever exhibits a valid committee signature on
an order that was not signed by the requisite number of honest nodes. By
\emph{(A3)}, $\Pr[\textsf{abort}] \le \epsilon_{\mathrm{sig}}$, so
$\mathbf{H}_1 \approx \mathbf{H}_0$. Conditioned on no abort we obtain the
invariant on which everything else rests: the order is finalised exactly
once, and---since an honest node releases its share only upon seeing a valid
signature on a finalised order---no honest share is published before
finalisation. \Adv{} holds at most $5 < 6$ shares of its own, so before
finalisation no ciphertext is decryptable, even though all $16$ nodes
(the $5$ corrupt ones included) received every bundle at broadcast time.

$\mathbf{H}_2$ replaces every honest node's decryption share by
$\mathsf{ShareSim}(\ct_\beta, \tx_\beta, \{sk_i\}_{i \in I})$. By
\emph{(A2)} this is statistically indistinguishable from $\mathbf{H}_1$.
This step must precede the next one: once share generation no longer
consults the honest secret keys, a reduction that does not know the
plaintext behind a ciphertext can still answer the share broadcast for it.

$\mathbf{H}_3^{i}$, for $i = 0, \dots, m$, encrypts $0^{|\tx_j|}$ in place
of $\tx_j$ for the first $i$ honest-user bundles and the real payload for
the remaining $m - i$; headers are unchanged throughout. We have

reduction receives the corrupt shares $\{sk_i\}_{i \in I}$ from the
threshold IND-CPA challenger, submits $(\tx_i, 0^{|\tx_i|})$ as its
challenge, embeds the challenge ciphertext at position $i$, and simulates
the rest of the execution unaided---by $\mathbf{H}_2$ the honest shares for
position $i$ are produced by $\mathsf{ShareSim}$ from $\tx_i$, which the
reduction learns only after $\pi$ is fixed, and the header $\hdr_i$ is part
of $\leak$ and hence available from the start. By \emph{(A1)} each step
costs at most $\epsilon_{\mathrm{cpa}}$.

bound of Theorem~\ref{thm:main}. Since \Adv{} fixes $\pi$ in $\mathbf{H}_1$
while its entire view is a function of $\leak$ and of $\mathbf{H}_3^{m}$'s
dummy ciphertexts, $\pi$ is independent of $\{\tx_\beta\}$ given
$\{\hdr_\beta\}$, which is the claimed frontrunning resistance.
\end{proof}

\subsection{Theorem}

We assume $\mathsf{TPKE}$ is a threshold KEM/DEM scheme: $ct=(c_1,c_2)$ with
$c_1$ a threshold-KEM encapsulation of a key $K$ and

shares are computed on $c_1$ alone. Let
$\mathbf{Adv}^{\mathsf{ow}}_{\mathsf{TKEM}}(\lambda,t)$ denote the advantage

\label{thm:main}
For every PPT adversary $\Adv$ statically corrupting a set $C$ of at most
$t=5$ committee members in the semi-honest model, together with a searcher
observing all public messages, there exists a PPT simulator $\Sim$ such that
for every environment $\mathcal{Z}$,
\[

  \;\approx_c\;
  \mathrm{EXEC}_{\Fencmp,\Sim,\mathcal{Z}}(\lambda).
\]
Concretely, for $q$ honest bundles per epoch and $q_H$ random-oracle queries,
the distinguishing advantage is at most
$q\cdot\mathbf{Adv}^{\mathsf{ow}}_{\mathsf{TKEM}}(\lambda,t)

\end{theorem}

\begin{proof}
$\Sim$ runs $\Adv$ internally and proceeds as follows.

\emph{Setup.} $\Sim$ executes $\mathsf{TPKE.KeyGen}$, publishes $cpk$, and
hands $\Adv$ the shares $\{sk_i\}_{i\in C}$. Corruption is static and $|C|\le
t$, so $\Sim$ knows exactly which shares $\Adv$ holds; it retains all $n$
shares itself.

\emph{Submission.} On $(\textsc{submitted},sid,\beta,(\hdr,\ell))$ from
$\Fencmp$, $\Sim$ samples a fresh KEM encapsulation $(c_1,K)$ honestly, sets
$c_2\leftarrow\{0,1\}^{\ell'}$ uniformly (with $\ell'$ the payload length
determined by $\ell$), and delivers the committee broadcast
$(\hdr,(c_1,c_2))$ to all $16$ simulated nodes --- in particular to the $5$
corrupt ones and to the searcher. \emph{This step is where $\hdr$ is
required}: without it in $\leak$, $\Sim$ cannot produce a broadcast whose
header is consistent with on-chain state, and the two executions separate.
For bundles submitted by corrupt users $\Sim$ knows $\pay$ outright and
follows the protocol.

\emph{Ordering.} $\Sim$ relays $\Adv$'s leader behaviour, obtains the proposed
sequence $\pi$, and sends $(\textsc{order},sid,\pi)$ to $\Fencmp$.

\emph{Reveal.} On $(\textsc{reveal},sid,\langle(\beta,b)\rangle)$, $\Sim$
learns each $\pay$. For every simulated bundle it programs
$H(K):=c_2\oplus\pay$, aborting if $H$ was already queried at $K$. It then
computes the $16$ decryption shares of $c_1$ honestly and broadcasts those of
the honest nodes; these recover the true $K$, so no equivocation on the
threshold component is needed, and the DEM layer now opens to the correct
$\pay$.

We argue indistinguishability through three hybrids.

\textbf{$\mathbf{H_0}$.} The real execution.

\textbf{$\mathbf{H_1}$.} Key material is generated by $\Sim$ rather than by
the dealer, and honest parties' internal state is computed by $\Sim$. With
static, semi-honest corruption this is a syntactic change: the joint
distribution of $(cpk,\{sk_i\}_{i\in C})$ is identical. $H_1\equiv H_0$.

\textbf{$\mathbf{H_2}$.} For each honest bundle in turn, replace
$c_2=\pay\oplus H(K)$ with a uniform string, deferring the definition of
$H(K)$ to reveal time. The two are identically distributed unless $\Adv$
queries $H$ at $K$ before reveal. Such a query yields $K$ from $c_1$ given
only the $t$ corrupt shares, so a distinguisher for the $j$-th step gives a
$\mathsf{TKEM}$ one-wayness adversary of the same running time. Summing over
the $q$ bundles and adding the $q\,q_H2^{-\lambda}$ probability that
programming aborts,
$|\Pr[H_2]-\Pr[H_1]|\le q\cdot\mathbf{Adv}^{\mathsf{ow}}_{\mathsf{TKEM}}
(\lambda,t)+q\,q_H2^{-\lambda}$.

\textbf{$\mathbf{H_3}$.} The ideal execution with $\Fencmp$ and $\Sim$. By
inspection, $H_2$ already consults the honest payloads only at reveal time,
and before that uses nothing beyond $(\hdr,\ell)$ per bundle --- exactly
$\leak$. Hence $H_3\equiv H_2$.

Chaining the three bounds gives the theorem. The threshold condition enters
only in $H_2$: with $|C|\le5<6$ the corrupt shares are insufficient to
reconstruct $K$, and every honest payload remains uniform in $\Adv$'s view
until $\pi$ is final.
\end{proof}
