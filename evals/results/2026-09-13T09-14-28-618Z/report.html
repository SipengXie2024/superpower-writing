def make_speedup_figure():
    base, base_sd = as_float(MEAN["revm"]), as_float(STD["revm"])
    x = np.arange(len(LABELS) + 1)          # last slot = geometric mean
    width = 0.36
    offsets = (np.arange(2) - 0.5) * width

    for (key, label), off in zip(SERIES[1:], offsets):
        means, errs = as_float(MEAN[key]), as_float(STD[key])
        sp = base / means
        # Error propagation for a ratio of two independent means.  This uses
        # the per-run std you reported; divide both terms by sqrt(5) if you
        # would rather plot the standard error of the mean.
        rel = np.sqrt((base_sd / base) ** 2 + (errs / means) ** 2)
